
    def get_ssl_certificate_info(self):
        """Get real SSL certificate expiration information"""
        try:
            import subprocess
            import datetime
            import sys
            
            # Debug logging
            debug_info = f"DEBUG SSL: CWD={os.getcwd()}, User={getpass.getuser()}"
            sys.stderr.write(debug_info + "\n")
            sys.stderr.flush()
            
            cmd = 'echo | openssl s_client -connect localhost:443 2>/dev/null | openssl x509 -noout -enddate'
            result = subprocess.run(cmd, shell=True, capture_output=True, text=True, timeout=10)
            
            sys.stderr.write(f"DEBUG SSL: Command result={result.returncode}, stdout={result.stdout[:50]}\n")
            sys.stderr.flush()
            
            if 'notAfter=' in result.stdout:
                date_str = result.stdout.split('notAfter=')[1].strip()
                exp_date = datetime.datetime.strptime(date_str, '%b %d %H:%M:%S %Y GMT')
                now = datetime.datetime.now()
                days_left = (exp_date - now).days
                sys.stderr.write(f"DEBUG SSL: Success! Days left={days_left}\n")
                sys.stderr.flush()
                return max(0, days_left)
            
            sys.stderr.write("DEBUG SSL: No notAfter found in output\n")
            sys.stderr.flush()
            return 89  # Fallback
        except Exception as e:
            import sys
            sys.stderr.write(f"DEBUG SSL: Exception={str(e)}\n")
            sys.stderr.flush()
            return 89  # Fallback
