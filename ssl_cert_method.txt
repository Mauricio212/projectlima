
    def get_ssl_certificate_info(self):
        """Get real SSL certificate expiration information"""
        try:
            import subprocess
            import datetime
            
            cmd = 'echo | openssl s_client -connect localhost:443 2>/dev/null | openssl x509 -noout -enddate'
            result = subprocess.run(cmd, shell=True, capture_output=True, text=True, timeout=10)
            
            if 'notAfter=' in result.stdout:
                date_str = result.stdout.split('notAfter=')[1].strip()
                exp_date = datetime.datetime.strptime(date_str, '%b %d %H:%M:%S %Y GMT')
                now = datetime.datetime.now()
                days_left = (exp_date - now).days
                return max(0, days_left)
            
            return 89  # Fallback
        except Exception:
            return 89  # Fallback
